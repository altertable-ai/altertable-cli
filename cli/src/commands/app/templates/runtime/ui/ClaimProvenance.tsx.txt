/** Sources, window, measures, and limits behind one claim. */
export type ClaimEvidence = {
  sources: string[];
  period: string;
  measures: string[];
  limitation?: string;
};

/** One app-authored finding derived from a displayed result. */
export type Claim = {
  id: string;
  headline: string;
  context?: string;
  evidence: ClaimEvidence;
  queryNames?: string[];
};
