import { formatCount } from "@altertable/data-app-runtime/format";
import type { Claim, ClaimEvidence } from "@altertable/data-app-runtime/ui";
import type { operations } from "./operations.ts";

type Tables = Awaited<ReturnType<typeof operations.tables.run>>;

export type TableClaim = Claim & { id: "visible" | "catalog" };

/**
 * Derive app-owned claims from the same result shown in the view. The operation caps
 * this list at 20, so these claims cannot describe the full catalog. Keep sources,
 * observation time, measures, and limitations with each claim.
 */
export function analyzeTables(tables: Tables, observedAt: string): TableClaim[] {
  function evidence(): ClaimEvidence {
    return {
      sources: ["information_schema.tables"],
      period: `Observed ${observedAt}`,
      measures: ["Tables shown"],
      limitation: "The query returns the first 20 visible tables, not the full catalog size.",
    };
  }
  const claims: TableClaim[] = [
    {
      id: "visible",
      headline: `${formatCount(tables.length)} tables in this view`,
      context: "These are the first visible tables ordered by catalog, schema, and name.",
      evidence: evidence(),
    },
  ];
  if (tables.length) {
    const firstCatalog = tables[0]!.catalog;
    const shown = tables.filter((table) => table.catalog === firstCatalog).length;
    claims.push({
      id: "catalog",
      headline: `${formatCount(shown)} shown from ${firstCatalog}`,
      context:
        "This count describes only the returned rows. Explore the full catalog before drawing broader conclusions.",
      evidence: evidence(),
    });
  }
  return claims;
}
