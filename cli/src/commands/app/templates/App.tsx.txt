import { createDataClient } from "@altertable/data-app-runtime/client";
import { createThemeController } from "@altertable/data-app-runtime/appearance";
import { formatCount } from "@altertable/data-app-runtime/format";
import { createDataHooks } from "@altertable/data-app-runtime/react";
import {
  AboutData,
  AppHeader,
  AppLayout,
  AppScope,
  AppToolbar,
  DataSection,
  SearchMatch,
  TableCard,
  ThemeSelector,
  resolveDataView,
  searchItems,
  useAppVariables,
} from "@altertable/data-app-runtime/ui";
import type { StoryStep } from "@altertable/data-app-runtime/ui";
import type { operations } from "./operations.ts";
import { dataContext } from "./data-context.ts";
import { variables, tableInput } from "./variables.ts";
import app from "../app.json";

const { useDataQuery } = createDataHooks(createDataClient<typeof operations>());
const theme = createThemeController(app.appearance);

type Table = Awaited<ReturnType<typeof operations.tables.run>>[number];

function tableList(rows: Table[]) {
  return (
    <ul className="table-list story-list">
      {rows.slice(0, 10).map((table) => (
        <li key={`${table.catalog}.${table.schema}.${table.name}`}>
          <strong>{table.name}</strong>
          <span>
            {table.catalog}.{table.schema}
          </span>
        </li>
      ))}
    </ul>
  );
}

/**
 * Assemble one app-owned view from a typed operation, presentation steps, and data context.
 * Runtime components handle request state, layout, explanation, and
 * presentation; this app chooses the content and when those controls appear.
 */
export function App() {
  const appVariables = useAppVariables(variables);
  const tables = useDataQuery("tables", tableInput(appVariables.values));
  const { tableSearch } = appVariables.values;
  const tableView = resolveDataView({
    requestedInput: "tables",
    current: tables.data ? { data: tables.data.data, input: "tables" } : undefined,
    pending: tables.isFetching,
    error: tables.error ?? undefined,
    sameInput: (left, right) => left === right,
    describe: () => "visible tables",
    isEmpty: () => false,
  });
  const rows = tables.data?.data ?? [];
  const storyEvidence: StoryStep["evidence"] = {
    sources: ["information_schema.tables"],
    period: `Observed ${new Date(tables.dataUpdatedAt).toISOString()}`,
    glossaryIds: ["tablesShown"],
    limitation: "These steps describe only the first 20 visible tables, not the full catalog.",
  };
  const steps: StoryStep[] = tables.data
    ? [
        {
          id: "visible",
          headline: `${formatCount(rows.length)} tables in this view`,
          context: "These are the first visible tables ordered by catalog, schema, and name.",
          evidence: storyEvidence,
          queries: tables.data.queries,
          queryNames: ["visible-tables"],
          visual: tableList(rows),
        },
        ...(rows.length
          ? [
              {
                id: "catalog",
                headline: `${formatCount(rows.filter((table) => table.catalog === rows[0]!.catalog).length)} shown from ${rows[0]!.catalog}`,
                context:
                  "This count describes only the returned rows. Explore the full catalog before drawing broader conclusions.",
                evidence: storyEvidence,
                queries: tables.data.queries,
                queryNames: ["visible-tables"],
                visual: tableList(rows.filter((table) => table.catalog === rows[0]!.catalog)),
              },
            ]
          : []),
      ]
    : [];

  return (
    <AppLayout footerActions={<ThemeSelector theme={theme} />}>
      <AppHeader
        scope={
          <AppScope organization={app.scope.organization} environment={app.scope.environment} />
        }
        title={app.title}
        description="Explore the tables available in this lakehouse."
        toolbar={
          <AppToolbar
            story={{
              title: app.title,
              scope: (
                <AppScope
                  organization={app.scope.organization}
                  environment={app.scope.environment}
                />
              ),
              dataContext,
              steps,
            }}
            aboutData={
              <AboutData
                id="data"
                shortcut
                dataContext={dataContext}
                queries={tables.data?.queries}
                iconOnly
                variant="elevated"
              />
            }
            refresh={{ refreshing: tables.isFetching, onRefresh: () => void tables.refetch() }}
          />
        }
      />

      <DataSection
        view={tableView}
        label="Available tables"
        error={{ title: "Couldn’t load tables", onRetry: () => void tables.refetch() }}
      >
        {(rows) => {
          const hits = searchItems(rows, tableSearch, {
            mode: "fuzzy",
            attributes: [
              { name: "name", getter: (table) => table.name, weight: 2 },
              { name: "schema", getter: (table) => table.schema },
              { name: "catalog", getter: (table) => table.catalog },
            ],
          });
          return (
            <TableCard
              title={`Available tables (${formatCount(rows.length)})`}
              description="Tables visible to the selected Altertable profile"
              columns={[
                {
                  id: "name",
                  header: "Table",
                  cell: (hit) => <SearchMatch match={hit.matches.name} />,
                },
                {
                  id: "schema",
                  header: "Schema",
                  cell: (hit) => <SearchMatch match={hit.matches.schema} />,
                },
                {
                  id: "catalog",
                  header: "Catalog",
                  cell: (hit) => <SearchMatch match={hit.matches.catalog} />,
                },
              ]}
              search={{
                itemCount: rows.length,
                label: "Search tables",
                value: tableSearch,
                onChange: (value) => appVariables.set("tableSearch", value),
              }}
              evidence={{
                id: "tables",
                glossaryEntry: dataContext.glossary.tablesShown,
                dataContext,
                queries: tables.data?.queries,
              }}
              empty={
                hits.length
                  ? undefined
                  : rows.length
                    ? {
                        title: "No matching tables",
                        description: "Try another search term.",
                      }
                    : {
                        title: "No visible tables",
                        description: "This profile has no lakehouse tables to explore yet.",
                      }
              }
              rows={hits}
              rowKey={({ item: table }) => `${table.catalog}.${table.schema}.${table.name}`}
            />
          );
        }}
      </DataSection>
    </AppLayout>
  );
}
