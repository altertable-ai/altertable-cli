import { readFile } from "node:fs/promises";
import { join } from "node:path";
import { pathToFileURL } from "node:url";

const appFile = (path) => pathToFileURL(join(process.cwd(), path)).href;

async function check() {
  const manifest = JSON.parse(await readFile("app.json", "utf8"));
  try {
    const { parseAppearance } = await import("@altertable/data-app/appearance");
    parseAppearance(manifest.appearance);
  } catch {
    throw new Error("app.json has invalid appearance settings.");
  }

  let registration;
  const hasQueries = await Bun.file("queries.json").exists();
  const hasVariables = await Bun.file("variables.json").exists();
  if (hasQueries || hasVariables) {
    if (!hasQueries || !hasVariables) throw new Error("Registered apps need both queries.json and variables.json.");
    const { defineDataAppRegistration } = await import("@altertable/data-app/contract");
    if (!defineDataAppRegistration) throw new Error("Upgrade @altertable/data-app to a release supporting separate query registration.");
    registration = defineDataAppRegistration({
      queries: JSON.parse(await readFile("queries.json", "utf8")),
      variables: JSON.parse(await readFile("variables.json", "utf8")),
    });
  }

  let operations;
  try {
    ({ operations } = await import(appFile("src/operations.ts")));
  } catch (error) {
    throw new Error(
      `Cannot load src/operations.ts: ${error instanceof Error ? error.message : "unknown error"}`,
    );
  }
  if (!operations || typeof operations !== "object" || Array.isArray(operations)) {
    throw new Error("src/operations.ts must export operations.");
  }
  const names = Object.keys(operations);
  if (!names.length) throw new Error("src/operations.ts needs at least one operation.");
  for (const name of names) {
    const operation = operations[name];
    if (
      !/^[a-z][a-z0-9-]*$/.test(name) ||
      !operation ||
      typeof operation.input !== "function" ||
      typeof operation.output !== "function" ||
      typeof operation.run !== "function" ||
      !Array.isArray(operation.checks) ||
      !operation.checks.length ||
      !Number.isInteger(operation.policy?.maxQueryRows) ||
      operation.policy.maxQueryRows < 1 ||
      !Number.isInteger(operation.policy.maxDurationMs) ||
      operation.policy.maxDurationMs < 1 ||
      (operation.policy.maxResponseBytes !== undefined &&
        (!Number.isInteger(operation.policy.maxResponseBytes) ||
          operation.policy.maxResponseBytes < 1))
    ) {
      throw new Error(`Operation "${name}" needs input and output parsers, check inputs, and limits.`);
    }
    for (const input of operation.checks) {
      try {
        operation.input(input);
      } catch {
        throw new Error(`Operation "${name}" has invalid check inputs.`);
      }
    }
  }

  if (await Bun.file("src/index.html").exists()) {
    const bundle = await Bun.build({
      entrypoints: ["src/index.html"],
      target: "browser",
      write: false,
      plugins: [{
        name: "data-app-browser-boundary",
        setup(build) {
          build.onResolve({ filter: /^@altertable\/data-app\/server(?:\/|$)/ }, () => {
            throw new Error("Browser code cannot import server transport. Use import type for server types.");
          });
          build.onLoad({ filter: /\.(?:[cm]?[jt]sx?|json)$/ }, ({ path }) => {
            const normalized = path.replaceAll("\\", "/");
            const root = process.cwd().replaceAll("\\", "/");
            if ((!registration && normalized === `${root}/src/operations.ts`) ||
                normalized === `${root}/queries.json` ||
                normalized === `${root}/src/server.ts` ||
                /node_modules\/@altertable\/data-app\/(?:dist\/(?:server|local)\.js|src\/server\/[^/]+\.[jt]s)$/.test(normalized)) {
              throw new Error("Browser code cannot import server operations or transport. Use import type for operation types.");
            }
          });
        },
      }],
    });
    if (!bundle.success) throw new Error(`Client import boundary failed: ${bundle.logs.join("\n")}`);
  }

  if (registration) {
    for (const operation of Object.values(operations)) {
      for (const id of Object.values(operation.queryNames ?? {})) {
        if (!Object.hasOwn(registration.queries, id)) throw new Error(`Unregistered query: ${id}`);
      }
    }
  }

  if (process.argv[1] !== "--lakehouse") return;
  const { createDataHandler } = await import("@altertable/data-app/server");
  const { localLakehouse } = await import("@altertable/data-app/server/bun");
  const lakehouse = localLakehouse(process.env);
  if (registration) {
    const { createRegisteredQueryHandler } = await import("@altertable/data-app/embed");
    const execute = createRegisteredQueryHandler(registration, async () => localLakehouse(process.env));
    lakehouse.queryRegistered = (operation, variables, { limit, signal }) => execute({ operation, variables, limit }, { signal });
    lakehouse.queryAll = async () => { throw new Error("Registered apps execute query IDs and values, not raw SQL."); };
  }
  const handle = createDataHandler(operations, async () => ({
    lakehouse,
    canDiscloseSql: true,
  }));
  for (const [name, operation] of Object.entries(operations)) {
    for (const input of operation.checks) {
      const response = await handle(
        new Request(`http://localhost/api/data/${name}`, {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify(input),
        }),
      );
      if (!response.ok)
        throw new Error(`Live check failed for operation "${name}" (${response.status}).`);
    }
  }
}

try {
  await check();
} catch (error) {
  console.error(error instanceof Error ? error.message : "Data app validation failed.");
  process.exitCode = 1;
}
