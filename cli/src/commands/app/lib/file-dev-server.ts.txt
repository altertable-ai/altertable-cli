import { watch } from "node:fs";
import { basename, dirname, isAbsolute, join } from "node:path";
import hostPage from "./host.html";
import { DataSourceError, sqlQueryRoute } from "@altertable/data-app/contract";
import { localLakehouse } from "@altertable/data-app/server/bun";

const appFile = process.env.ALTERTABLE_APP_FILE;
if (typeof appFile !== "string" || appFile.length === 0) {
  throw new Error("Missing data app file.");
}
const port = Number(process.env.PORT);
if (!process.env.PORT || !Number.isInteger(port)) throw new Error("Missing preview port.");

const lakehouse = localLakehouse();
const clients = new Set();
let reloadTimer;

function scheduleReload() {
  clearTimeout(reloadTimer);
  reloadTimer = setTimeout(() => {
    console.log("Reloading data app.");
    for (const client of clients) {
      try {
        client.enqueue("data: reload\n\n");
      } catch {
        clients.delete(client);
      }
    }
  }, 50);
}

try {
  watch(dirname(appFile), (_, filename) => {
    const name = filename == null ? undefined : filename.toString();
    if (name === undefined || name === basename(appFile)) scheduleReload();
  });
} catch (error) {
  console.error(
    `Watching ${appFile} failed: ${error instanceof Error ? error.message : "unknown error"}.`,
  );
}

function runtimePackages() {
  const from = join(import.meta.dir, "package.json");
  const installed = `${join(import.meta.dir, "node_modules")}/`.replaceAll("\\", "/");
  return {
    name: "altertable-file-dev-packages",
    setup(build) {
      // The app file lives outside this runtime. Resolve its package imports here,
      // then let Bun resolve dependencies from the installed package tree.
      build.onResolve({ filter: /^[^./]/ }, (args) => {
        if (args.namespace !== "file" || args.importer.replaceAll("\\", "/").startsWith(installed))
          return;
        if (args.path.startsWith("node:") || args.path.startsWith("bun:")) return;
        try {
          const resolved = Bun.resolveSync(args.path, from);
          return isAbsolute(resolved) ? { path: resolved } : undefined;
        } catch {
          return undefined;
        }
      });
    },
  };
}

async function bundleApp() {
  const result = await Bun.build({
    entrypoints: [appFile],
    target: "browser",
    format: "iife",
    // Prefer browser entry points. Bun's default conditions include "node" and
    // would pull Node builds such as fflate's createRequire entry.
    conditions: ["browser", "import", "module", "default"],
    define: { "import.meta.env": "{}" },
    plugins: [runtimePackages()],
  });
  if (!result.success) {
    const details = result.logs
      .map((log) => {
        const notes = (log.notes ?? []).map((note) => note.text).filter(Boolean);
        return [log.message, ...notes].filter(Boolean).join("\n");
      })
      .filter(Boolean)
      .join("\n");
    throw new Error(details || "Bundling the data app failed.");
  }
  const scripts = result.outputs.filter((output) => output.path.endsWith(".js"));
  if (scripts.length !== 1) throw new Error("The data app must bundle to one script.");
  const javascript = await scripts[0].text();
  const stylesheet = result.outputs.find((output) => output.path.endsWith(".css"));
  if (!stylesheet) return javascript;
  const css = await stylesheet.text();
  return `const style = document.createElement("style"); style.textContent = ${JSON.stringify(css)}; document.head.append(style);\n${javascript}`;
}

function readQuery(body) {
  try {
    return sqlQueryRoute.input(JSON.parse(body));
  } catch {
    return undefined;
  }
}

// Queries run with the profile's credentials. The Host check defeats DNS rebinding; the Origin
// check rejects cross-site POSTs, which browsers send without a CORS preflight.
function localHosts(server) {
  return [`127.0.0.1:${server.port}`, `localhost:${server.port}`];
}

function isLocalHost(request, server) {
  return localHosts(server).includes(request.headers.get("host"));
}

function isSameOrigin(request, server) {
  const site = request.headers.get("sec-fetch-site");
  const origin = request.headers.get("origin");
  return (
    (site === null || site === "same-origin") &&
    localHosts(server).some((host) => origin === `http://${host}`)
  );
}

function forbidden() {
  return new Response("Forbidden", { status: 403 });
}

const server = Bun.serve({
  hostname: "127.0.0.1",
  port,
  idleTimeout: 60,
  routes: { "/": hostPage },
  async fetch(request, server) {
    if (!isLocalHost(request, server)) return forbidden();
    const url = new URL(request.url);
    if (url.pathname === "/bundle.js") {
      try {
        return new Response(await bundleApp(), {
          headers: {
            "content-type": "text/javascript; charset=utf-8",
            "cache-control": "no-store",
          },
        });
      } catch (error) {
        const message = error instanceof Error ? error.message : "Bundling the data app failed.";
        console.error(message);
        return new Response(message, {
          status: 500,
          headers: { "content-type": "text/plain; charset=utf-8", "cache-control": "no-store" },
        });
      }
    }
    if (url.pathname === "/__dev/reload") {
      server.timeout(request, 0);
      return new Response(
        new ReadableStream({
          start(controller) {
            clients.add(controller);
            controller.enqueue("event: ready\ndata: ok\n\n");
            request.signal.addEventListener("abort", () => clients.delete(controller));
          },
        }),
        { headers: { "content-type": "text/event-stream", "cache-control": "no-cache" } },
      );
    }
    if (url.pathname === "/api/sql" && request.method === "POST") {
      if (!isSameOrigin(request, server)) return forbidden();
      const body = await request.text();
      if (body.length > 100_000) return new Response("Query too large", { status: 413 });
      const query = readQuery(body);
      if (!query) return new Response("Invalid query", { status: 400 });
      try {
        const result = await lakehouse.queryAll(query.statement, {
          limit: query.limit,
          signal: request.signal,
          ...(query.params === undefined ? {} : { params: query.params }),
        });
        return Response.json({
          columns: result.columns,
          rows: result.rows,
          ...(result.queryId === undefined ? {} : { queryId: result.queryId }),
        });
      } catch (error) {
        const reason = error instanceof DataSourceError ? error.reason : "unavailable";
        console.error(
          `Data app query failed (${reason}): ${error instanceof Error ? error.message : reason}.`,
        );
        return Response.json({ reason }, { status: 502 });
      }
    }
    return new Response("Not found", { status: 404 });
  },
});

console.log(`Data app running at ${server.url} (${appFile})`);
